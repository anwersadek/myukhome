function doPost(e){var s=SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data');if(!s)s=SpreadsheetApp.getActiveSpreadsheet().insertSheet('Data');var d=JSON.parse(e.postData.contents);s.clear();s.getRange(1,1).setValue(JSON.stringify(d));s.getRange(1,2).setValue(new Date().toISOString());return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON)}function doGet(e){var s=SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data');if(!s)return ContentService.createTextOutput(JSON.stringify({status:'empty'})).setMimeType(ContentService.MimeType.JSON);var v=s.getRange(1,1).getValue();var t=s.getRange(1,2).getValue();return ContentService.createTextOutput(JSON.stringify({status:'ok',data:JSON.parse(v),updated:t})).setMimeType(ContentService.MimeType.JSON)}
3
Click Deploy → New deployment → Web app → set "Who has access" to Anyone → Deploy → Copy the URL and paste below
